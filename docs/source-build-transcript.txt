Important:

NEXT_PUBLIC_SUPABASE_ANON_KEY is safe for browser.
SUPABASE_SERVICE_ROLE_KEY must only be used on the server.
Never expose service-role keys in client components.

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

exception when duplicate_object then null; end $$;

create table if not exists public.connection_requests (
  id uuid primary key default gen_random_uuid(),
  from_user uuid not null references public.profiles(id) on delete cascade,
  to_user uuid not null references public.profiles(id) on delete cascade,
  status public.connection_status not null default 'pending',
  note text,

  updated_at timestamptz not null default now(),
  decided_at timestamptz,
  unique(from_user, to_user)
);

create index if not exists idx_connection_requests_from on public.connection_requests(from_user);
create index if not exists idx_connection_requests_to on public.connection_requests(to_user);

create or replace trigger set_connection_requests_updated_at
before update on public.connection_requests
for each row execute function public.set_updated_at();

alter table public.connection_requests enable row level security;

drop policy if exists "Users can view own connection requests" on public.connection_requests;
create policy "Users can view own connection requests"
on public.connection_requests
for select
to authenticated
using (from_user = auth.uid() or to_user = auth.uid());

drop policy if exists "Users can send connection requests" on public.connection_requests;
create policy "Users can send connection requests"
on public.connection_requests
for insert
to authenticated
with check (
  from_user = auth.uid()
  and to_user is distinct from auth.uid()
);

drop policy if exists "Users can manage own connection requests" on public.connection_requests;
create policy "Users can manage own connection requests"
on public.connection_requests
for update
to authenticated
using (
  (from_user = auth.uid() and status = 'pending')
  or
  (to_user = auth.uid() and status = 'pending')
)
with check (
  (from_user = auth.uid() and status in ('pending','cancelled'))
  or
  (to_user = auth.uid() and status in ('pending','accepted','rejected'))
);

import { createAdminClient } from '@/lib/supabase/admin';

type CandidateProfile = {

  full_name: string;
  date_of_birth: string | null;
  city: string | null;
  bio: string | null;

export type CompatibilityMatch = {
  id: string;
  full_name: string;
  age: number | null;
  city: string | null;
  bio: string | null;
  avatar_url: string | null;
  score: number;
  dimension_scores: Record<string, number>;
  highlights: string[];
  sharedInterests: string[];
  goals: string[];
  interests: string[];
};

const DIMENSIONS = [
  'mindset',
  'personality',
  'life_goals',
  'interests',
  'communication',
  'social_expectations',
] as const;

const WEIGHTS: Record<(typeof DIMENSIONS)[number], number> = {
  mindset: 0.25,
  personality: 0.2,
  life_goals: 0.2,
  interests: 0.15,
  communication: 0.1,
  social_expectations: 0.1,
};

function normalize(value: string) {
  return value.trim().toLowerCase();
}

function ageFromDob(dob: string | null) {
  if (!dob) return null;

  const birth = new Date(dob);
  if (Number.isNaN(birth.getTime())) return null;

  const today = new Date();
  let age = today.getFullYear() - birth.getFullYear();
  const monthDiff = today.getMonth() - birth.getMonth();

  if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
    age--;
  }

  return age;
}

function overlapScore(a: string[], b: string[]) {
  const listA = a.map(normalize).filter(Boolean);
  const listB = b.map(normalize).filter(Boolean);

  if (!listA.length || !listB.length) return 15;

  const shared = listA.filter((item) => listB.includes(item));

  if (!shared.length) return 15;

  const ratio = shared.length / Math.min(listA.length, listB.length);
  return Math.round(40 + 60 * Math.min(1, ratio));
}

function tagOverlapScore(a: string[], b: string[]) {
  const listA = a.map(normalize).filter(Boolean);
  const listB = b.map(normalize).filter(Boolean);

  if (!listA.length || !listB.length) return 15;

  const shared = listA.filter((item) => listB.includes(item));

  if (!shared.length) return 15;

  const ratio = shared.length / Math.min(listA.length, listB.length);
  return Math.round(40 + 60 * Math.min(1, ratio));
}

function buildDimensionTags(
  answers: AssessmentAnswers,
  questions: AssessmentQuestion[],
  tagMap: Map<number, Map<string, string[]>>
) {
  const result: Record<string, string[]> = {};

  for (const dimension of DIMENSIONS) {
    if (dimension === 'interests') {
      result[dimension] = [];
      continue;
    }

    const tagSet = new Set<string>();

    for (const question of questions) {
      if (question.dimension !== dimension) continue;

      const selectedValues = answers.get(question.id) || [];
      const valueTagMap = tagMap.get(question.id) || new Map();

      for (const value of selectedValues) {
        const tags = valueTagMap.get(value) || [];
        tags.forEach((tag) => tagSet.add(normalize(tag)));
      }
    }

    result[dimension] = Array.from(tagSet);
  }

  return result;
}

function dimensionPersonalityScore(aTags: string[], bTags: string[], baseScore: number) {
  const a = new Set(aTags.map(normalize));
  const b = new Set(bTags.map(normalize));

  const complementaryPairs: [string, string][] = [
    ['introvert', 'balanced'],
    ['extrovert', 'balanced'],
    ['analytical', 'social'],
    ['private', 'friends'],
    ['steady', 'energy'],
  ];

  let isComplementary = false;

  for (const [x, y] of complementaryPairs) {
    if ((a.has(x) && b.has(y)) || (a.has(y) && b.has(x))) {
      isComplementary = true;
      break;
    }
  }

  if (isComplementary && baseScore < 78) {
    return 78;
  }

  return baseScore;
}

export async function getCompatibilityMatches(userId: string): Promise<CompatibilityMatch[]> {
  const admin = createAdminClient();

  const { data: ownProfile } = await admin
    .from('profiles')
    .select('id, goals, is_discoverable, assessment_completed')
    .eq('id', userId)
    .maybeSingle();

  if (!ownProfile || !ownProfile.assessment_completed || !ownProfile.is_discoverable) {
    return [];
  }

  const { data: ownInterests } = await admin
    .from('user_interests')
    .select('name')
    .eq('profile_id', userId);

  const { data: ownRun } = await admin
    .from('assessment_runs')
    .select('assessment_answers(question_id, selected_values)')
    .eq('profile_id', userId)
    .maybeSingle();

  if (!ownRun) return [];

  const { data: questions } = await admin
    .from('assessment_questions')
    .select('*')
    .eq('is_active', true)
    .order('sort_order');

  if (!questions) return [];

  const ownAnswers: AssessmentAnswers = new Map(
    (ownRun.assessment_answers || []).map((answer) => [
      answer.question_id,
      answer.selected_values || [],
    ])
  );

  const { data: setting } = await admin
    .from('app_settings')
    .select('value')
    .eq('key', 'compatibility_threshold')
    .maybeSingle();

  const threshold = Number((setting?.value as unknown as string) || '80');

  const { data: candidates } = await admin
    .from('profiles')
    .select(
      `
      id,
      full_name,
      date_of_birth,
      city,
      bio,
      avatar_url,
      goals,
      show_age,
      show_city,
      user_interests(name)
      `
    )
    .eq('assessment_completed', true)
    .eq('is_discoverable', true)
    .neq('id', userId)
    .limit(200);

  if (!candidates || candidates.length === 0) return [];

  const { data: blockedRows } = await admin
    .from('blocked_users')
    .select('blocker_id, blocked_id')
    .or(`blocker_id.eq.${userId},blocked_id.eq.${userId}`);

  const blockedSet = new Set(
    (blockedRows || [])
      .flatMap((row) => [row.blocker_id, row.blocked_id])
      .filter((id) => id && id !== userId)
  );

  const { data: connections } = await admin
    .from('connections')
    .select('user_a, user_b')
    .or(`user_a.eq.${userId},user_b.eq.${userId}`);

  const connectedSet = new Set(
    (connections || [])
      .map((row) => (row.user_a === userId ? row.user_b : row.user_a))
      .filter(Boolean)
  );

  const { data: dismissed } = await admin
    .from('dismissed_recommendations')
    .select('target_id')
    .eq('user_id', userId);

  const dismissedSet = new Set((dismissed || []).map((row) => row.target_id));

  const { data: rejectedRequests } = await admin
    .from('connection_requests')
    .select('from_user, to_user')
    .eq('status', 'rejected')
    .or(`from_user.eq.${userId},to_user.eq.${userId}`);

  const rejectedSet = new Set(
    (rejectedRequests || [])
      .flatMap((row) => [row.from_user, row.to_user])
      .filter((id) => id && id !== userId)
  );

  const { data: pendingRequests } = await admin
    .from('connection_requests')
    .select('from_user, to_user')
    .eq('status', 'pending')
    .or(`from_user.eq.${userId},to_user.eq.${userId}`);

  const pendingSet = new Set(
    (pendingRequests || [])
      .flatMap((row) => [row.from_user, row.to_user])
      .filter((id) => id && id !== userId)
  );

  const eligibleCandidates = (candidates as CandidateProfile[]).filter((candidate) => {
    const age = ageFromDob(candidate.date_of_birth);

    if (age !== null && age < 18) return false;
    if (blockedSet.has(candidate.id)) return false;
    if (connectedSet.has(candidate.id)) return false;
    if (dismissedSet.has(candidate.id)) return false;
    if (rejectedSet.has(candidate.id)) return false;
    if (pendingSet.has(candidate.id)) return false;

    return true;
  });

  if (eligibleCandidates.length === 0) return [];

  const candidateIds = eligibleCandidates.map((candidate) => candidate.id);

  const { data: candidateRuns } = await admin
    .from('assessment_runs')
    .select('profile_id, assessment_answers(question_id, selected_values)')
    .in('profile_id', candidateIds);

  const answersByProfile = new Map<string, AssessmentAnswers>();

  for (const run of candidateRuns || []) {
    const answers: AssessmentAnswers = new Map(
      (run.assessment_answers || []).map((answer) => [
        answer.question_id,
        answer.selected_values || [],
      ])
    );

    answersByProfile.set(run.profile_id, answers);
  }

  const tagMap = new Map<number, Map<string, string[]>>();

  for (const question of questions as AssessmentQuestion[]) {
    const valueMap = new Map<string, string[]>();

    for (const option of question.options || []) {
      valueMap.set(option.value, option.tags || []);
    }

    tagMap.set(question.id, valueMap);
  }

  const ownDimensionTags = buildDimensionTags(
    ownAnswers,
    questions as AssessmentQuestion[],
    tagMap
  );

  const ownInterestsNormalized = (ownInterests || []).map(
    (interest) => normalize(interest.name)
  );

  const matches: CompatibilityMatch[] = [];
  const toUpsert: {
    user_a: string;
    user_b: string;
    score: number;
    dimension_scores: Record<string, number>;
    highlights: string[];
  }[] = [];

  for (const candidate of eligibleCandidates) {
    const candidateAnswers = answersByProfile.get(candidate.id);
    if (!candidateAnswers) continue;

    const candidateDimensionTags = buildDimensionTags(
      candidateAnswers,
      questions as AssessmentQuestion[],
      tagMap
    );

    const candidateInterests = (candidate.user_interests || []).map(
      (interest) => normalize(interest.name)
    );

    const sharedInterests = candidateInterests.filter((interest) =>
      ownInterestsNormalized.includes(interest)
    );

    const dimension_scores: Record<string, number> = {};
    let weightedTotal = 0;

    for (const dimension of DIMENSIONS) {
      let score: number;

      if (dimension === 'interests') {
        score = overlapScore(
          (ownInterests || []).map((interest) => interest.name),
          (candidate.user_interests || []).map((interest) => interest.name)
        );
      } else if (dimension === 'life_goals') {
        const tagScore = tagOverlapScore(
          ownDimensionTags.life_goals || [],
          candidateDimensionTags.life_goals || []
        );

        const goalScore = overlapScore(
          (ownProfile.goals || []) as string[],
          (candidate.goals || []) as string[]
        );

        score = Math.round(tagScore * 0.65 + goalScore * 0.35);
      } else {
        const baseScore = tagOverlapScore(
          ownDimensionTags[dimension] || [],
          candidateDimensionTags[dimension] || []
        );

        if (dimension === 'personality') {
          score = dimensionPersonalityScore(
            ownDimensionTags.personality || [],
            candidateDimensionTags.personality || [],
            baseScore
          );
        } else {
          score = baseScore;
        }
      }

      dimension_scores[dimension] = score;
      weightedTotal += score * WEIGHTS[dimension];
    }

    const score = Math.round(Math.max(0, Math.min(100, weightedTotal)));

    if (score < threshold) continue;

    const highlights: string[] = [];

    if (sharedInterests.length > 0) {
      highlights.push(
        `Shared interests: ${sharedInterests.slice(0, 2).map((item) => item.charAt(0).toUpperCase() + item.slice(1)).join(', ')}`
      );
    }

    if ((dimension_scores.mindset || 0) >= 85) highlights.push('Similar mindset');
    if ((dimension_scores.personality || 0) >= 85) highlights.push('Complementary personality');
    if ((dimension_scores.life_goals || 0) >= 85) highlights.push('Shared life goals');
    if ((dimension_scores.communication || 0) >= 85) highlights.push('Similar communication preferences');
    if ((dimension_scores.social_expectations || 0) >= 85) highlights.push('Similar social expectations');

    const age = ageFromDob(candidate.date_of_birth);

    matches.push({
      id: candidate.id,
      full_name: candidate.full_name,
      age: candidate.show_age ? age : null,
      city: candidate.show_city ? candidate.city : null,
      bio: candidate.bio,
      avatar_url: candidate.avatar_url,
      score,
      dimension_scores,
      highlights: highlights.slice(0, 4),
      sharedInterests: sharedInterests.slice(0, 4),
      goals: (candidate.goals || []).slice(0, 4),
      interests: (candidate.user_interests || []).map((interest) => interest.name).slice(0, 6),
    });

    const pair = [userId, candidate.id].sort();

    toUpsert.push({
      user_a: pair[0],
      user_b: pair[1],
      score,
      dimension_scores,
      highlights,
    });
  }

  if (toUpsert.length > 0) {
    await admin
      .from('compatibility_scores')
      .upsert(toUpsert, { onConflict: 'user_a,user_b' });
  }

  return matches.sort((a, b) => b.score - a.score);
}

function splitList(value: FormDataEntryValue | null) {

export async function completeProfile(formData: FormData) {

  const city = String(formData.get('city') || '').trim();
  const bio = String(formData.get('bio') || '').trim();

  const showCity = String(formData.get('show_city') || 'false') === 'true';

  if (!fullName || !username || !dateOfBirth) {
    redirect('/onboarding?error=missing_fields');
  }

  const avatarEntry = formData.get('avatar');
  let avatarUrl: string | null = null;

  if (avatarEntry instanceof File && avatarEntry.size > 0) {
    const extension = avatarEntry.name.split('.').pop() || 'jpg';
    const path = `${user.id}/avatar-${Date.now()}.${extension}`;

    const arrayBuffer = await avatarEntry.arrayBuffer();

    await admin.storage
      .from('connect-photos')
      .upload(path, arrayBuffer, { contentType: avatarEntry.type || 'image/jpeg' });

    const { data: publicData } = admin.storage
      .from('connect-photos')
      .getPublicUrl(path);

    avatarUrl = publicData.publicUrl;
  }

  await admin.from('profiles').upsert(
    {
      id: user.id,
      full_name: fullName,
      username,
      date_of_birth: dateOfBirth,
      gender: gender || null,
      city: city || null,
      bio: bio || null,

      show_age: showAge,
      show_city: showCity,
      avatar_url: avatarUrl,
      assessment_completed: false,
      is_discoverable: true,
    },
    { onConflict: 'id' }
  );

  await admin.from('user_interests').delete().eq('profile_id', user.id);

  if (interests.length > 0) {
    await admin
      .from('user_interests')
      .insert(
        interests.map((name) => ({
          profile_id: user.id,
          name,
        }))

export async function submitAssessment(payload: AssessmentPayload[]) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();

  if (!user) {
    return { error: 'You must be signed in.' };
  }

  if (!payload || payload.length === 0) {
    return { error: 'Please answer all questions.' };
  }

  const admin = createAdminClient();

  const { data: existingRun } = await admin
    .from('assessment_runs')
    .select('id, version')
    .eq('profile_id', user.id)
    .maybeSingle();

  let runId: string;

  if (existingRun) {
    runId = existingRun.id;

    await admin
      .from('assessment_runs')
      .update({
        status: 'submitted',
        version: (existingRun.version || 1) + 1,
        submitted_at: new Date().toISOString(),
      })
      .eq('profile_id', user.id);
  } else {
    const { data: createdRun } = await admin
      .from('assessment_runs')
      .insert({
        profile_id: user.id,
        status: 'submitted',
        version: 1,
        submitted_at: new Date().toISOString(),
      })
      .select('id')
      .single();

    runId = createdRun?.id as string;
  }

  if (!runId) {
    return { error: 'Unable to save assessment run.' };
  }

  await admin.from('assessment_answers').delete().eq('run_id', runId);

  await admin
    .from('assessment_answers')
    .insert(
      payload.map((item) => ({
        run_id: runId,
        question_id: item.questionId,
        selected_values: item.selectedValues,
      }))
    );

  await admin
    .from('profiles')
    .update({ assessment_completed: true })
    .eq('id', user.id);

  try {
    await getCompatibilityMatches(user.id);
  } catch {
    // Compatibility recalculation can be retried on next discover visit.
  }

  revalidatePath('/discover');
  revalidatePath('/requests');
  revalidatePath('/profile');

  return { ok: true };
}

function normalizePair(a: string, b: string): [string, string] {
  return a < b ? [a, b] : [b, a];
}

export async function sendConnectionRequest(targetId: string, note?: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();

  if (!user) return { error: 'You must be signed in.' };
  if (!targetId) return { error: 'Missing target user.' };
  if (targetId === user.id) return { error: 'You cannot connect with yourself.' };

  const admin = createAdminClient();

  const { data: existingRequest } = await admin
    .from('connection_requests')
    .select('id')
    .or(`and(from_user.eq.${user.id},to_user.eq.${targetId}),and(from_user.eq.${targetId},to_user.eq.${user.id})`)
    .maybeSingle();

  if (existingRequest) {
    return { error: 'A connection request already exists.' };
  }

  const { data: existingConnection } = await admin
    .from('connections')
    .select('id')
    .or(
      `and(user_a.eq.${user.id},user_b.eq.${targetId}),and(user_a.eq.${targetId},user_b.eq.${user.id})`
    )
    .maybeSingle();

  if (existingConnection) {
    return { error: 'You are already connected.' };
  }

  const { data: blockA } = await admin
    .from('blocked_users')
    .select('id')
    .eq('blocker_id', user.id)
    .eq('blocked_id', targetId)
    .maybeSingle();

  const { data: blockB } = await admin
    .from('blocked_users')
    .select('id')
    .eq('blocker_id', targetId)
    .eq('blocked_id', user.id)
    .maybeSingle();

  if (blockA || blockB) {
    return { error: 'You cannot send a connection request right now.' };
  }

  await admin.from('connection_requests').insert({
    from_user: user.id,
    to_user: targetId,
    status: 'pending',
    note: note || null,
  });

  await admin.from('notifications').insert({
    user_id: targetId,
    type: 'connection_request',
    actor_id: user.id,
    payload: { from_user: user.id },
  });

  revalidatePath('/discover');
  revalidatePath('/requests');
  revalidatePath('/notifications');

  return { ok: true };
}

export async function acceptConnection(requestId: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();

  if (!user) return { error: 'You must be signed in.' };

  const admin = createAdminClient();

  const { data: request } = await admin
    .from('connection_requests')
    .select('id, from_user, to_user, status')
    .eq('id', requestId)
    .maybeSingle();

  if (!request) return { error: 'Request not found.' };
  if (request.to_user !== user.id) return { error: 'Not allowed.' };
  if (request.status !== 'pending') return { error: 'Request already handled.' };

  await admin
    .from('connection_requests')
    .update({
      status: 'accepted',
      decided_at: new Date().toISOString(),
    })
    .eq('id', requestId);

  const [userA, userB] = normalizePair(request.from_user, request.to_user);

  const { data: connection, error: connectionError } = await admin
    .from('connections')
    .insert({ user_a: userA, user_b: userB })
    .select('id')
    .maybeSingle();

  if (!connectionError && connection) {
    const { data: conversation } = await admin
      .from('conversations')
      .insert({ type: 'private' })
      .select('id')
      .single();

    if (conversation) {
      await admin.from('conversation_members').insert([
        { conversation_id: conversation.id, user_id: request.from_user },
        { conversation_id: conversation.id, user_id: request.to_user },
      ]);
    }
  }

  await admin.from('notifications').insert([
    {
      user_id: request.from_user,
      type: 'connection_accepted',
      actor_id: user.id,
      payload: { to_user: user.id },
    },
    {
      user_id: request.to_user,
      type: 'system',
      actor_id: user.id,
      payload: { connection_accepted: true },
    },
  ]);

  revalidatePath('/requests');
  revalidatePath('/chats');
  revalidatePath('/discover');
  revalidatePath('/notifications');

  return { ok: true };
}

'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { useTransition } from 'react';
import { submitAssessment } from '@/actions/assessment';

type QuestionOption = {
  value: string;
  label: string;
};

type Question = {
  id: number;
  prompt: string;
  answer_type: 'single' | 'multi';
  options: QuestionOption[];
};

export function AssessmentClient({
  questions,
  initialAnswers,
}: {
  questions: Question[];
  initialAnswers: Record<number, string[]>;
}) {
  const router = useRouter();
  const [index, setIndex] = useState(0);
  const [answers, setAnswers] = useState<Record<number, string[]>>(initialAnswers);
  const [isPending, startTransition] = useTransition();

  const question = questions[index];

  if (!question) {
    return null;
  }

  function toggleOption(value: string) {
    const current = answers[question.id] || [];

    if (question.answer_type === 'single') {
      setAnswers({ ...answers, [question.id]: [value] });
      return;
    }

    if (current.includes(value)) {
      setAnswers({
        ...answers,
        [question.id]: current.filter((item) => item !== value),
      });
    } else {
      setAnswers({
        ...answers,
        [question.id]: [...current, value],
      });
    }
  }

  function goNext() {
    if (index < questions.length - 1) {
      setIndex(index + 1);
    }
  }

  function goBack() {
    if (index > 0) {
      setIndex(index - 1);
    }
  }

  function submit() {
    startTransition(async () => {
      const payload = questions
        .map((q) => ({
          questionId: q.id,
          selectedValues: answers[q.id] || [],
        }))
        .filter((item) => item.selectedValues.length > 0);

      const result = await submitAssessment(payload);

      if (result.error) {
        alert(result.error);
        return;
      }

      router.push('/discover');
      router.refresh();
    });
  }

  return (
    <div className="flex min-h-screen items-center justify-center bg-zinc-50 p-6">
      <div className="w-full max-w-xl rounded-2xl border border-zinc-200 bg-white p-6 shadow-sm">
        <div className="mb-6 flex items-center justify-between text-sm text-zinc-500">
          <span>CONNECT assessment</span>
          <span>
            {index + 1}/{questions.length}
          </span>
        </div>

        <div className="mb-4 h-2 w-full overflow-hidden rounded-full bg-zinc-100">
          <div
            className="h-full bg-zinc-900 transition-all"
            style={{
              width: `${((index + 1) / questions.length) * 100}%`,
            }}
          />
        </div>

        <h1 className="text-xl font-semibold">{question.prompt}</h1>
        <p className="mt-1 text-sm text-zinc-500">
          {question.answer_type === 'single'
            ? 'Choose one option.'
            : 'Choose all that apply.'}
        </p>

        <div className="mt-6 space-y-3">
          {question.options.map((option) => {
            const selected = (answers[question.id] || []).includes(option.value);

            return (
              <button
                key={option.value}
                type="button"
                onClick={() => toggleOption(option.value)}
                className={
                  selected
                    ? 'w-full rounded-xl border border-zinc-900 bg-zinc-900 px-4 py-3 text-left text-white'
                    : 'w-full rounded-xl border border-zinc-200 px-4 py-3 text-left hover:bg-zinc-50'
                }
              >
                {option.label}
              </button>
            );
          })}
        </div>

        <div className="mt-8 flex items-center justify-between">
          <button
            type="button"
            onClick={goBack}
            disabled={index === 0 || isPending}
            className="rounded-xl px-4 py-2 text-sm text-zinc-600 hover:text-zinc-900 disabled:opacity-40"
          >
            Back
          </button>

          {index < questions.length - 1 ? (
            <button
              type="button"
              onClick={goNext}
              disabled={isPending}
              className="rounded-xl bg-zinc-900 px-5 py-2 text-sm text-white disabled:opacity-60"
            >
              Next
            </button>
          ) : (
            <button
              type="button"
              onClick={submit}
              disabled={isPending}
              className="rounded-xl bg-zinc-900 px-5 py-2 text-sm text-white disabled:opacity-60"
            >
              {isPending ? 'Saving...' : 'Finish'}
            </button>
          )}
        </div>
      </div>
    </div>
  );
}

  const { data: { user } } = await supabase.auth.getUser();

  if (!user) redirect('/login');

  const { data: incoming } = await supabase
    .from('connection_requests')
    .select('id, from_user, to_user, status, note, created_at')
    .eq('to_user', user.id)
    .eq('status', 'pending')
    .order('created_at', { ascending: false });

  const { data: outgoing } = await supabase
    .from('connection_requests')
    .select('id, from_user, to_user, status, note, created_at')
    .eq('from_user', user.id)
    .neq('status', 'cancelled')
    .order('created_at', { ascending: false });

  const { data: conns } = await supabase
    .from('connections')
    .select('user_a, user_b')
    .or(`user_a.eq.${user.id},user_b.eq.${user.id}`);

  const otherIds = new Set<string>();
  (incoming || []).forEach((r) => otherIds.add(r.from_user));
  (outgoing || []).forEach((r) => otherIds.add(r.to_user));
  (conns || []).forEach((c) => otherIds.add(c.user_a === user.id ? c.user_b : c.user_a));
  otherIds.delete(user.id);

  const { data: profiles } = await supabase
    .from('profiles')
    .select('id, full_name, username, avatar_url, city, show_city, bio')
    .in('id', Array.from(otherIds));
  const profileById = new Map((profiles || []).map((p) => [p.id, p]));

  // Build a map: otherUserId -> private conversation id
  const { data: myConvMembers } = await supabase
    .from('conversation_members')
    .select('conversation_id, user_id')
    .eq('user_id', user.id);
  const myConvIds = (myConvMembers || []).map((m) => m.conversation_id);

  let convByOther = new Map<string, string>();
  if (myConvIds.length) {
    const { data: privConvs } = await supabase
      .from('conversations')
      .select('id')
      .in('id', myConvIds)
      .eq('type', 'private');
    const privIds = (privConvs || []).map((c) => c.id);
    if (privIds.length) {
      const { data: privMembers } = await supabase
        .from('conversation_members')
        .select('conversation_id, user_id')
        .in('conversation_id', privIds);
      const byConv = new Map<string, string[]>();
      (privMembers || []).forEach((m) => {
        const a = byConv.get(m.conversation_id) || [];
        a.push(m.user_id);
        byConv.set(m.conversation_id, a);
      });
      for (const [convId, members] of byConv) {
        if (members.includes(user.id)) {
          const other = members.find((m) => m !== user.id);
          if (other) convByOther.set(other, convId);
        }
      }
    }
  }

  const incomingList = (incoming || [])
    .map((r) => ({ id: r.id, note: r.note, created_at: r.created_at, profile: profileById.get(r.from_user) }))
    .filter((r) => r.profile);

  const outgoingList = (outgoing || [])
    .map((r) => ({ id: r.id, status: r.status, created_at: r.created_at, profile: profileById.get(r.to_user) }))
    .filter((r) => r.profile);

  const connections = (conns || [])
    .map((c) => {
      const other = c.user_a === user.id ? c.user_b : c.user_a;
      const p = profileById.get(other);
      return { id: other, profile: p, conversationId: convByOther.get(other) || null };
    })
    .filter((c) => c.profile);

  return (
    <RequireAuth>
      <RequestsClient
        incoming={incomingList}
        outgoing={outgoingList}
        connections={connections}
      />
    </RequireAuth>
  );

import { createServerSupabase } from '@/lib/supabase/server';

export type ConversationListItem = {
  id: string;
  type: 'private' | 'group';
  title: string | null;
  group_id: string | null;
  otherProfile?: { id: string; full_name: string; avatar_url: string | null };
  groupInfo?: { id: string; name: string; icon_url: string | null };
  memberProfiles: { id: string; full_name: string; avatar_url: string | null }[];
  lastMessage?: {
    id: string;
    content: string | null;
    media_url: string | null;
    sender_id: string | null;
    created_at: string;
  };
  unreadCount: number;
};

export async function getConversationList(userId: string): Promise<ConversationListItem[]> {
  const supabase = await createServerSupabase();

  const { data: memberships } = await supabase
    .from('conversation_members')
    .select('conversation_id, last_read_at')
    .eq('user_id', userId);

  if (!memberships || memberships.length === 0) return [];

  const conversationIds = memberships.map((m) => m.conversation_id);
  const lastReadByConversation = new Map(memberships.map((m) => [m.conversation_id, m.last_read_at]));

  const { data: conversations } = await supabase
    .from('conversations')
    .select('id, type, group_id, title')
    .in('id', conversationIds);

  if (!conversations || conversations.length === 0) return [];

  const { data: messages } = await supabase
    .from('messages')
    .select('id, conversation_id, sender_id, content, media_url, created_at')
    .in('conversation_id', conversationIds)
    .order('created_at', { ascending: true });

  const messagesByConversation = new Map<string, NonNullable<typeof messages>>();
  for (const m of messages || []) {
    const arr = messagesByConversation.get(m.conversation_id) || [];
    arr.push(m);
    messagesByConversation.set(m.conversation_id, arr);
  }

  const { data: allMembers } = await supabase
    .from('conversation_members')
    .select('conversation_id, user_id')
    .in('conversation_id', conversationIds);

  const userIds = Array.from(new Set((allMembers || []).map((m) => m.user_id)));
  const { data: profiles } = await supabase
    .from('profiles')
    .select('id, full_name, avatar_url')
    .in('id', userIds);
  const profileById = new Map((profiles || []).map((p) => [p.id, p]));

  const groupIds = Array.from(new Set((conversations || []).map((c) => c.group_id).filter(Boolean)));
  let groups: { id: string; name: string; icon_url: string | null }[] = [];
  if (groupIds.length > 0) {
    const { data: g } = await supabase
      .from('groups')
      .select('id, name, icon_url')
      .in('id', groupIds);
    groups = g || [];
  }
  const groupById = new Map(groups.map((g) => [g.id, g]));

  const membersByConversation = new Map<string, string[]>();
  for (const m of allMembers || []) {
    const arr = membersByConversation.get(m.conversation_id) || [];
    arr.push(m.user_id);
    membersByConversation.set(m.conversation_id, arr);
  }

  return conversations.map((c) => {
    const msgs = messagesByConversation.get(c.id) || [];
    const lastMessage = msgs.length ? msgs[msgs.length - 1] : undefined;
    const lastRead = lastReadByConversation.get(c.id) || null;
    const unreadCount = msgs.filter(
      (m) => m.sender_id !== userId && (!lastRead || m.created_at > lastRead)
    ).length;

    const memberIds = membersByConversation.get(c.id) || [];
    const memberProfiles = memberIds
      .map((id) => profileById.get(id))
      .filter(Boolean)
      .map((p) => ({ id: p!.id, full_name: p!.full_name, avatar_url: p!.avatar_url }));

    const item: ConversationListItem = {
      id: c.id,
      type: c.type,
      title: c.title,
      group_id: c.group_id,
      memberProfiles,
      lastMessage,
      unreadCount,
    };

    if (c.type === 'private') {
      const otherId = memberIds.find((id) => id !== userId);
      const other = otherId ? profileById.get(otherId) : undefined;
      if (other) item.otherProfile = { id: other.id, full_name: other.full_name, avatar_url: other.avatar_url };
    } else if (c.group_id) {
      const g = groupById.get(c.group_id);
      if (g) item.groupInfo = { id: g.id, name: g.name, icon_url: g.icon_url };
    }

    return item;
  });
}

'use server';

import { createServerSupabase } from '@/lib/supabase/server';
import { createAdminClient } from '@/lib/supabase/admin';

export async function ensurePrivateConversation(otherUserId: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };
  if (!otherUserId || otherUserId === user.id) return { error: 'Invalid user' };

  const admin = createAdminClient();

  const { data: conn } = await admin
    .from('connections')
    .select('id')
    .or(`and(user_a.eq.${user.id},user_b.eq.${otherUserId}),and(user_a.eq.${otherUserId},user_b.eq.${user.id})`)
    .maybeSingle();
  if (!conn) return { error: 'You are not connected with this user.' };

  const { data: myMemberships } = await admin
    .from('conversation_members')
    .select('conversation_id')
    .eq('user_id', user.id);
  const myConvIds = (myMemberships || []).map((m) => m.conversation_id);

  let conversationId: string | null = null;
  if (myConvIds.length) {
    const { data: convs } = await admin
      .from('conversations')
      .select('id, type')
      .in('id', myConvIds)
      .eq('type', 'private');
    for (const c of convs || []) {
      const { data: otherMember } = await admin
        .from('conversation_members')
        .select('user_id')
        .eq('conversation_id', c.id)
        .eq('user_id', otherUserId)
        .maybeSingle();
      if (otherMember) {
        conversationId = c.id;
        break;
      }
    }
  }

  if (!conversationId) {
    const { data: newConv } = await admin
      .from('conversations')
      .insert({ type: 'private' })
      .select('id')
      .single();
    conversationId = newConv?.id || null;
    if (conversationId) {
      await admin.from('conversation_members').insert([
        { conversation_id: conversationId, user_id: user.id },
        { conversation_id: conversationId, user_id: otherUserId },
      ]);
    }
  }

  if (!conversationId) return { error: 'Could not start conversation.' };
  return { ok: true, conversationId };
}

export async function sendMessage(conversationId: string, content: string, mediaUrl?: string) {

type Message = {

function time(ts: string) {
  return new Date(ts).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
}

function ChatWindow({
  conversation,
  currentUserId,
}: {
  conversation: ConversationListItem;
  currentUserId: string;
}) {
  const supabase = useMemo(() => createBrowserSupabase(), []);
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState('');
  const [typing, setTyping] = useState(false);
  const [loading, setLoading] = useState(true);
  const channelRef = useRef<ReturnType<typeof supabase.channel> | null>(null);
  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
  const lastTypingSent = useRef(0);
  const fileRef = useRef<HTMLInputElement>(null);
  const bottomRef = useRef<HTMLDivElement>(null);

  const nameById = useMemo(() => {
    const m = new Map<string, string>();
    conversation.memberProfiles.forEach((p) => m.set(p.id, p.full_name));
    return m;
  }, [conversation]);

  const title =
    conversation.type === 'group'
      ? conversation.groupInfo?.name || 'Group'
      : conversation.otherProfile?.full_name || 'Chat';

  useEffect(() => {
    let mounted = true;

    async function init() {
      const { data } = await supabase
        .from('messages')
        .select('id, conversation_id, sender_id, content, media_url, created_at')
        .eq('conversation_id', conversation.id)
        .order('created_at', { ascending: true });
      if (mounted) {
        setMessages(data || []);
        setLoading(false);
        markConversationRead(conversation.id);
      }
    }
    init();

    const channel = supabase
      .channel(`chat-${conversation.id}`)
      .on(
        'postgres_changes',
        { event: 'INSERT', schema: 'public', table: 'messages', filter: `conversation_id=eq.${conversation.id}` },
        (payload) => {
          const m = payload.new as Message;
          setMessages((prev) => (prev.some((x) => x.id === m.id) ? prev : [...prev, m]));
          if (m.sender_id !== currentUserId) markConversationRead(conversation.id);
        }
      )
      .on('broadcast', { event: 'typing' }, (payload) => {
        if (payload.payload?.userId === currentUserId) return;
        setTyping(true);
        if (typingTimeout.current) clearTimeout(typingTimeout.current);
        typingTimeout.current = setTimeout(() => setTyping(false), 2500);
      })
      .subscribe();
    channelRef.current = channel;

    return () => {
      mounted = false;
      if (typingTimeout.current) clearTimeout(typingTimeout.current);
      supabase.removeChannel(channel);
    };
  }, [conversation.id, currentUserId, supabase]);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages, typing]);

  function emitTyping() {
    const now = Date.now();
    if (now - lastTypingSent.current < 1500) return;
    lastTypingSent.current = now;
    channelRef.current?.send({
      type: 'broadcast',
      event: 'typing',
      payload: { userId: currentUserId },
    });
  }

  async function sendText() {
    if (!input.trim()) return;
    const result = await sendMessage(conversation.id, input);
    if (result.ok && result.message) {
      setMessages((prev) =>
        prev.some((x) => x.id === result.message.id) ? prev : [...prev, result.message]
      );
    }
    setInput('');
    channelRef.current?.send({
      type: 'broadcast',
      event: 'typing',
      payload: { userId: currentUserId, done: true },
    });
  }

  async function sendPhoto(file: File) {

      return;
    }
    const { data } = supabase.storage.from('connect-photos').getPublicUrl(path);
    const result = await sendMessage(conversation.id, '', data.publicUrl);
    if (result.ok && result.message) {
      setMessages((prev) => [...prev, result.message]);
    }
  }

  return (
    <div className="flex h-[70vh] min-h-[420px] flex-col rounded-2xl border border-zinc-200 bg-white">

            <Users size={18} className="text-zinc-500" />
          ) : conversation.otherProfile?.avatar_url ? (
            // eslint-disable-next-line @next/next/no-img-element
            <img src={conversation.otherProfile.avatar_url} alt={title} className="h-full w-full object-cover" />
          ) : (
            <User size={18} className="text-zinc-500" />
          )}
        </div>
        <div>
          <div className="font-medium">{title}</div>
          <div className="text-xs text-zinc-500">
            {typing ? (
              <span className="italic text-zinc-700">typing…</span>
            ) : (
              conversation.type === 'group'
                ? `${conversation.memberProfiles.length} members`
                : 'Connected'
            )}
          </div>
        </div>

      <div className="flex-1 space-y-3 overflow-y-auto p-4">
        {loading ? (

        ) : messages.length === 0 ? (
          <div className="flex h-full items-center justify-center text-sm text-zinc-400">
            No messages yet. Say hello 👋
          </div>
        ) : (
          messages.map((m) => {
            const mine = m.sender_id === currentUserId;
            return (

import { revalidatePath } from 'next/cache';

export async function createGroup(formData: FormData) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };

  const admin = createAdminClient();

  const name = String(formData.get('name') || '').trim();
  const description = String(formData.get('description') || '').trim();
  const category = String(formData.get('category') || '').trim();
  const visibility = String(formData.get('visibility') || 'public');
  const membershipPolicy = String(formData.get('membership_policy') || 'open');

  if (!name) return { error: 'Group name is required.' };

  const slug = name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '');

  const { data: group, error } = await admin
    .from('groups')
    .insert({
      name,
      description: description || null,
      category: category || null,
      visibility: visibility === 'private' ? 'private' : 'public',
      membership_policy: membershipPolicy === 'request' ? 'request' : 'open',
      created_by: user.id,
      slug: slug || null,
    })
    .select('id')
    .single();

  if (error || !group) return { error: 'Failed to create group.' };

  const { data: conv } = await admin
    .from('conversations')
    .insert({ type: 'group', group_id: group.id, title: name })
    .select('id')
    .single();

  await admin.from('group_members').insert({
    group_id: group.id,
    user_id: user.id,
    role: 'owner',
    status: 'approved',
  });

  if (conv) {
    await admin.from('conversation_members').insert({ conversation_id: conv.id, user_id: user.id });
  }

  revalidatePath('/groups');
  return { ok: true, groupId: group.id, conversationId: conv?.id };
}

export async function joinGroup(groupId: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };

  const admin = createAdminClient();

  const { data: group } = await admin
    .from('groups')
    .select('id, membership_policy, name')
    .eq('id', groupId)
    .maybeSingle();
  if (!group) return { error: 'Group not found.' };

  const { data: existing } = await admin
    .from('group_members')
    .select('status')
    .eq('group_id', groupId)
    .eq('user_id', user.id)
    .maybeSingle();
  if (existing) return { error: 'You are already in this group.' };

  const status = group.membership_policy === 'request' ? 'pending' : 'approved';

  await admin.from('group_members').insert({
    group_id: groupId,
    user_id: user.id,
    role: 'member',
    status,
  });

  if (status === 'approved') {
    const { data: conv } = await admin
      .from('conversations')
      .select('id')
      .eq('group_id', groupId)
      .eq('type', 'group')
      .maybeSingle();
    if (conv) {
      await admin.from('conversation_members').insert({ conversation_id: conv.id, user_id: user.id });
    }
  }

  revalidatePath('/groups');
  return { ok: true, status };
}

  if (conv) {

}

                    >
                      <LogOut size={16} /> Leave
                    </button>
                  </>
                ) : g.status === 'pending' ? (
                  <span className="text-sm text-zinc-500">Membership pending approval…</span>
                ) : (

import { createServerSupabase } from '@/lib/supabase/server';

export type PhotoPost = {
  id: string;
  profile_id: string;
  caption: string | null;
  created_at: string;
  author: { id: string; full_name: string; username: string; avatar_url: string | null };
  media: { id: number; url: string; sort_order: number }[];
  likes: number;
  likedByMe: boolean;
  comments: {
    id: string;
    content: string;
    created_at: string;
    profile_id: string;

  }[];
};

export async function createPhotoPost(formData: FormData) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };

  const admin = createAdminClient();
  const caption = String(formData.get('caption') || '').trim();
  const files = formData.getAll('photos');

  if (!files.length) return { error: 'Please select at least one photo.' };

  const { data: post, error: postError } = await admin
    .from('photo_posts')
    .insert({ profile_id: user.id, caption: caption || null })
    .select('id')
    .single();

  if (postError || !post) return { error: 'Failed to create post.' };

  const mediaUploads = [];
  for (let i = 0; i < files.length; i++) {
    const file = files[i];
    if (!file || !(file instanceof File)) continue;

    const ext = file.name.split('.').pop() || 'jpg';
    const path = `${user.id}/posts/${post.id}/${i}-${Date.now()}.${ext}`;
    const arrayBuffer = await file.arrayBuffer();

    const { error: uploadError } = await admin.storage
      .from('connect-photos')
      .upload(path, arrayBuffer, { contentType: file.type || 'image/jpeg' });

    if (uploadError) continue;

    const { data: publicData } = admin.storage.from('connect-photos').getPublicUrl(path);
    mediaUploads.push({ post_id: post.id, url: publicData.publicUrl, sort_order: i });
  }

  if (mediaUploads.length > 0) {
    await admin.from('photo_post_media').insert(mediaUploads);
  } else {
    await admin.from('photo_posts').delete().eq('id', post.id);
    return { error: 'All photo uploads failed.' };
  }

  revalidatePath('/photos');
  return { ok: true, postId: post.id };
}

export async function deletePhotoPost(postId: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };

  const admin = createAdminClient();
  const { data: post } = await admin
    .from('photo_posts')
    .select('id, profile_id')
    .eq('id', postId)
    .maybeSingle();

  if (!post || post.profile_id !== user.id) return { error: 'Not authorized.' };

  const { data: media } = await admin
    .from('photo_post_media')
    .select('url')
    .eq('post_id', postId);

  if (media) {
    for (const m of media) {
      const path = m.url.split('/connect-photos/')[1];
      if (path) await admin.storage.from('connect-photos').remove([path]);
    }
  }

  await admin.from('photo_post_media').delete().eq('post_id', postId);
  await admin.from('photo_posts').delete().eq('id', postId);
  revalidatePath('/photos');
  return { ok: true };
}

export async function toggleLike(postId: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };

  const admin = createAdminClient();
  const { data: existing } = await admin
    .from('photo_likes')
    .select('id')
    .eq('post_id', postId)
    .eq('profile_id', user.id)
    .maybeSingle();

  if (existing) {
    await admin.from('photo_likes').delete().eq('id', existing.id);
    return { ok: true, liked: false };
  }

  await admin.from('photo_likes').insert({ post_id: postId, profile_id: user.id });
  return { ok: true, liked: true };
}

export async function addComment(postId: string, content: string) {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Not authenticated' };
  if (!content.trim()) return { error: 'Comment is empty.' };

  const admin = createAdminClient();
  const { data: post } = await admin
    .from('photo_posts')
    .select('id, profile_id')
    .eq('id', postId)
    .maybeSingle();
  if (!post) return { error: 'Post not found.' };

  await admin.from('photo_comments').insert({ post_id: postId, profile_id: user.id, content: content.trim() });

  if (post.profile_id !== user.id) {
    await admin.from('notifications').insert({
      user_id: post.profile_id,
      type: 'photo_comment',
      actor_id: user.id,
      payload: { post_id: postId },
    });
  }

  revalidatePath('/photos');
  return { ok: true };
}

export async function getPhotoFeed(page = 1, limit = 20): Promise<{ posts: PhotoPost[]; hasMore: boolean }> {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { posts: [], hasMore: false };

  const offset = (page - 1) * limit;

  const { data: connections } = await supabase
    .from('connections')
    .select('user_a, user_b')
    .or(`user_a.eq.${user.id},user_b.eq.${user.id}`);

  const connectedIds = new Set<string>();
  connectedIds.add(user.id);
  (connections || []).forEach(c => connectedIds.add(c.user_a === user.id ? c.user_b : c.user_a));

  const { data: posts, count } = await supabase
    .from('photo_posts')
    .select('id, profile_id, caption, created_at', { count: 'exact' })
    .in('profile_id', Array.from(connectedIds))
    .order('created_at', { ascending: false })
    .range(offset, offset + limit - 1);

  if (!posts) return { posts: [], hasMore: false };

  const postIds = posts.map(p => p.id);
  const userIds = Array.from(new Set(posts.map(p => p.profile_id)));

  const { data: profiles } = await supabase
    .from('profiles')
    .select('id, full_name, username, avatar_url')
    .in('id', userIds);
  const profileById = new Map((profiles || []).map(p => [p.id, p]));

  const { data: media } = await supabase
    .from('photo_post_media')
    .select('post_id, id, url, sort_order')
    .in('post_id', postIds);
  const mediaByPost = new Map<string, typeof media>();
  (media || []).forEach(m => { const a = mediaByPost.get(m.post_id) || []; a.push(m); mediaByPost.set(m.post_id, a); });

  const { data: likes } = await supabase
    .from('photo_likes')
    .select('post_id, profile_id')
    .in('post_id', postIds);
  const likesByPost = new Map<string, Set<string>>();
  (likes || []).forEach(l => { const s = likesByPost.get(l.post_id) || new Set(); s.add(l.profile_id); likesByPost.set(l.post_id, s); });

  const { data: comments } = await supabase
    .from('photo_comments')
    .select('post_id, id, content, created_at, profile_id')
    .in('post_id', postIds)
    .order('created_at', { ascending: false })
    .limit(100);
  const commentsByPost = new Map<string, typeof comments>();
  (comments || []).forEach(c => { const a = commentsByPost.get(c.post_id) || []; a.push(c); commentsByPost.set(c.post_id, a); });

  const commentUserIds = Array.from(new Set((comments || []).map(c => c.profile_id)));
  const { data: cProfiles } = await supabase
    .from('profiles')
    .select('id, full_name, avatar_url')
    .in('id', commentUserIds);
  const cProfileById = new Map((cProfiles || []).map(p => [p.id, p]));

  return {
    posts: posts.map(p => ({
      id: p.id,
      profile_id: p.profile_id,
      caption: p.caption,
      created_at: p.created_at,
      author: profileById.get(p.profile_id) || { id: p.profile_id, full_name: 'Unknown', username: '', avatar_url: null },
      media: (mediaByPost.get(p.id) || [])
        .sort((a, b) => a.sort_order - b.sort_order)
        .map(m => ({ id: m.id, url: m.url, sort_order: m.sort_order })),
      likes: (likesByPost.get(p.id) || new Set()).size,
      likedByMe: (likesByPost.get(p.id) || new Set()).has(user.id),
      comments: (commentsByPost.get(p.id) || []).map(c => ({
        id: c.id,
        content: c.content,
        created_at: c.created_at,
        profile_id: c.profile_id,
        full_name: cProfileById.get(c.profile_id)?.full_name || 'Unknown',
        avatar_url: cProfileById.get(c.profile_id)?.avatar_url || null,
      })),
    })),
    hasMore: page * limit < (count || 0),
  };
}

  actor_id: string | null;
  payload: Record<string, any>;
  read: boolean;
  created_at: string;
  actor?: { full_name: string; avatar_url: string | null };
};

export async function getNotifications(page = 1, limit = 20): Promise<{ notifications: Notification[]; hasMore: boolean }> {
  const supabase = await createServerSupabase();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { notifications: [], hasMore: false };

  const offset = (page - 1) * limit;
  const { data: rows, count } = await supabase
    .from('notifications')
    .select('id, type, actor_id, payload, read, created_at', { count: 'exact' })
    .eq('user_id', user.id)
    .order('created_at', { ascending: false })
    .range(offset, offset + limit - 1);

  if (!rows) return { notifications: [], hasMore: false };

  const actorIds = Array.from(new Set((rows || []).map(r => r.actor_id).filter(Boolean)));
  const { data: actors } = await supabase
    .from('profiles')
    .select('id, full_name, avatar_url')
    .in('id', actorIds);

  const actorById = new Map((actors || []).map(a => [a.id, a]));

  return {
    notifications: rows.map(r => ({
      ...r,
      actor: r.actor_id ? actorById.get(r.actor_id) || undefined : undefined,
    })),
    hasMore: page * limit < (count || 0),
  };
}

function getIcon(type: string) {
  switch (type) {
    case 'connection_request': case 'connection_accepted': return <User size={16} />;
    case 'message': case 'group_message': return <MessageCircle size={16} />;
    case 'group_invitation': case 'group_approval': return <Users size={16} />;
    case 'photo_like': case 'photo_comment': return <Heart size={16} />;
    default: return <User size={16} />;
  }
}

function timeAgo(ts: string) {
  const diff = Date.now() - new Date(ts).getTime();
  const mins = Math.floor(diff / 60000);
  if (mins < 60) return `${mins}m ago`;
  const hrs = Math.floor(mins / 60);
  if (hrs < 24) return `${hrs}h ago`;
  return `${Math.floor(hrs / 24)}d ago`;
}

export function NotificationsClient({ currentUserId, initialNotifications }: { currentUserId: string; initialNotifications: Notification[] }) {
  const router = useRouter();
  const [isPending, startTransition] = useTransition();
  const [list, setList] = useState(initialNotifications);

  function run(fn: () => Promise<unknown>) {
    startTransition(async () => { await fn(); router.refresh(); });
  }

  function handleRead(id: string) {
    setList(prev => prev.map(n => n.id === id ? { ...n, read: true } : n));
    run(() => markRead(id));
  }

  function handleMarkAll() {
    setList(prev => prev.map(n => ({ ...n, read: true })));
    run(() => markAllRead());
  }

  return (
    <div className="p-4 md:p-8">
      <div className="mb-6 flex items-center justify-between">
        <h1 className="text-2xl font-semibold">Notifications</h1>
        <button onClick={handleMarkAll} className="text-sm text-zinc-600 hover:text-zinc-900">Mark all read</button>
      </div>

      <div className="space-y-2">
        {list.length === 0 ? (
          <div className="rounded-2xl border border-dashed border-zinc-300 p-10 text-center text-sm text-zinc-500">No notifications yet.</div>
        ) : (
          list.map(n => (
            <div key={n.id} className={`flex items-center gap-3 rounded-xl border border-zinc-200 p-4 ${!n.read ? 'bg-zinc-50' : 'bg-white'}`}>
              <div className="flex h-10 w-10 items-center justify-center rounded-full bg-zinc-100 text-zinc-600">
                {getIcon(n.type)}
              </div>
              <div className="flex-1">
                <div className="text-sm">
                  {n.actor ? <span className="font-medium">{n.actor.full_name}</span> : 'System'}
                  {' '}
                  <span className="text-zinc-600">
                    {n.type.replace(/_/g, ' ').replace(/\b\w/g, c => c.toUpperCase())}
                  </span>
                </div>
                <div className="text-xs text-zinc-400">{timeAgo(n.created_at)}</div>
              </div>
              {!n.read && <button onClick={() => handleRead(n.id)} className="text-xs text-blue-600 hover:underline">Mark read</button>}

src/components/error-boundary.tsx

"use client";

import { Component, ErrorInfo, ReactNode } from "react";
import { initSentry } from "@/lib/sentry";

interface Props { children: ReactNode }
interface State { hasError: boolean }

export class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(): State { return { hasError: true }; }

  componentDidCatch(error: Error, info: ErrorInfo) {
    initSentry();
    console.error("[ErrorBoundary]", error, info);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="flex min-h-screen items-center justify-center bg-background p-6 text-center">
          <div>
            <h1 className="text-2xl font-semibold text-foreground">Something went wrong</h1>
            <p className="mt-2 text-muted-foreground">We&apos;ve been notified and are fixing it.</p>
            <button onClick={() => window.location.reload()} className="mt-4 rounded-xl bg-brand-600 px-4 py-2 text-white">
              Reload app
            </button>
          </div>
        </div>
      );
    }
    return this.props.children;
  }
}
